// How many of ONE arm's three wrist mounts are live. Paired with a plain
// on/off per arm rather than folded into it -- see CVARINFO: the owner asked
// for a switch and a count, so that flicking the arm off does not lose the
// count and changing the count does not switch the arm on.
//
// REPLACES "RS_HardPointArmActiveCount" (Off / Forearm only / Wrist only /
// All six). The forearm tier is deferred as a feature, so its rows are gone
// rather than sitting in the menu doing nothing; the wrist tier is now three

	1, "1"
	2, "2"
	3, "3"

// The arm-anchored wrist hardpoints -- three around each wrist, six total.
// SafeCommand rather than Command for the destructive/modal ones: edit mode
// and reset both change state you would not want triggered by a misread
// controller press while scrolling a menu in VR.
OptionMenu "RS_HardPointOptions"
{
	Title "Arm Hardpoints"
	StaticText " "

	// EXACTLY THE SHAPE THE OWNER SPECIFIED (2026-08-26): a switch and a
	// count, per hand, and nothing else. Three wrist mounts per arm, six
	// total -- the same six anchors the single-arm layout had, re-laid out
	// rather than added to.
	//
	// NO FOREARM ROWS. That tier is deferred as a feature, not cancelled
	// (it was going to hold utility items -- flares, shields, usable
	// inventory -- and the content model for that is not built). Everything
	// learned about its placement is preserved in the DEFERRED block on
	// RS_HardPoints.zs' constants; what is NOT preserved is a menu row for
	// something that does nothing.
	StaticText "Mounts -- MAIN hand", "Gold"
	Option "Wrist slots",       "rs_hardpoint_wrist_main", "OnOff"
	Option "How many",          "rs_hardpoint_wrist_main_count", "RS_HardPointWristCount"
	StaticText " "

	StaticText "Mounts -- OFF hand", "Gold"
	Option "Wrist slots",       "rs_hardpoint_wrist_off", "OnOff"
	Option "How many",          "rs_hardpoint_wrist_off_count", "RS_HardPointWristCount"
	StaticText " "

	StaticText "Placement", "Gold"
	SafeCommand "Edit Mode -- grab spheres and move them", "netevent rs-hardpoint-edit"
	SafeCommand "Print tuned table to console",            "netevent rs-hardpoint-table"
	SafeCommand "Reset offsets to defaults",               "netevent rs-hardpoint-reset"
	// ADDED 2026-08-26. NetworkProcess has handled this netevent since the
	// split, with no alias and no menu row anywhere -- reachable only by
	// typing it into the console.
	SafeCommand "Recalibrate standing height",             "netevent rs-hardpoint-recalibrate"
	StaticText " "

	// The CURRENT edit-table -- tune it with Edit Mode above, then save it,
	// so the tuning survives a restart without a rebuild. Loading takes
	// effect immediately. One slot only: RS_Holsters keeps a seated and a
	// standing profile because its anchors hang off a body, and these are
	// bolted to a tracked controller, where posture is not a variable.
	//
	// ONE SLOT still covers BOTH ARMS. Each arm has its own layout -- the
	// mounts sit relative to whichever controller wears them, so one shared
	// table could not serve both -- but the two live in separate key ranges
	// inside the same file ("m0_".."m2_" and "o0_".."o2_", see profileKey),
	// so saving writes both arms and loading restores both. A layout saved
	// before 2026-08-26 uses the old flat "h0_".."h5_" keys and will simply
	// not be found; every field falls back to the new defaults, which is the
	// honest outcome given indices 0-2 used to mean the forearm row.
	StaticText "Saved Layout", "Gold"
	SafeCommand "Save tuned layout", "netevent rs-hardpoint-save-layout"
	SafeCommand "Load tuned layout", "netevent rs-hardpoint-load-layout"
	StaticText " "

	StaticText "Appearance", "Gold"
	Option "Show hardpoint markers",    "rs_hardpoint_markers", "OnOff"

	Option "Marker shape",              "rs_hardpoint_marker_shape", "RS_HardPointMarkerShape"
	Option "Marker color (cold)",       "rs_hardpoint_marker_color", "RS_HardPointMarkerColor"
	Option "Marker color (hot)",        "rs_hardpoint_marker_color_hot", "RS_HardPointMarkerColor"
	Slider "Marker scale",              "rs_hardpoint_marker_scale", 0.3, 3.0, 0.1, 2
	Slider "Hardpoint item size (fallback)", "rs_hardpoint_prop_scale_arm", 0.02, 0.5, 0.01, 2
	// The torso-scale dial. Every mount in this fork is arm-anchored, so it
	// is the slider above that does the work here -- but this cvar was READ
	// by RS_HardPointProp.holsterPropScale() while being declared in no
	// CVARINFO at all, which meant it silently ran at its hardcoded 0.18
	// fallback with no way to see or change it. Declared and given a row

	Slider "Fill amount (measured models)",  "rs_hardpoint_prop_fill", 0.1, 1.0, 0.05, 2

	Slider "Stored item pitch",         "rs_hardpoint_prop_pitch", -180.0, 180.0, 5.0, 0

	Slider "Stored item up/down",       "rs_hardpoint_prop_up",  -12.0, 12.0, 0.5, 1
	Slider "Stored item fwd/back",      "rs_hardpoint_prop_fwd", -12.0, 12.0, 0.5, 1
	Slider "Stored item side",          "rs_hardpoint_prop_side", -12.0, 12.0, 0.5, 1
	StaticText " "

	StaticText "Gesture-Cast Arming", "Gold"
	Option "Enable gesture-cast",       "rs_hardpoint_gesture_enable", "OnOff"

	Slider "Palm-out roll tolerance",   "rs_hardpoint_gesture_roll_tolerance", 5.0, 90.0, 5.0, 0
	StaticText " "

	StaticText "Timing & Sound", "Gold"
	Option "Instant store/draw",       "rs_hardpoint_instant_switch", "OnOff"
	Option "Store/draw confirm sound", "rs_hardpoint_sound", "OnOff"
	Option "Confirm sound style",      "rs_hardpoint_sound_style", "RS_HardPointSoundStyle"
	StaticText " "

	// ENGINE cvar (DoomXR), not this pk3's own -- referencing it here does
	// not require this pk3 to own or declare it. Two handed weapons /
	// stabilize distance used to be listed here too, but those are general
	// VR control settings, not anything hardpoint-specific -- they already

	// here was just two ways to find the same setting.
